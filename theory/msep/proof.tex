For an integer $M\ge2$ let $\Sig_{\le M}$ be the class of $\Orb\in\Sig$ whose cone orders are all at most $M$ (surfaces included).

\begin{theorem}[Bounded cone orders]\label{msep:thm}
Let $M\ge2$.
\begin{enumerate}[label=(\roman*)]
\item If $\Orb,\Orb'\in\Sig_{\le M}$ and $H_M(\Orb)=H_M(\Orb')$, then $\sigma(\Orb)=\sigma(\Orb')$. Hence $\Kmult(\Orb;\Sig_{\le M})\le\min\big(M,\lfloor\Area(\Orb)/\pi\rfloor+4\big)$ for every $\Orb\in\Sig_{\le M}$.
\item The number $M$ cannot be lowered. Put $w_a=\prod_{1\le b\le M,\,b\ne a}(a^2-b^2)^{-1}$, let $C$ be the least positive integer for which $\nu(a)=C\,a\,w_a$ is an integer for $2\le a\le M$ and $s=\sum_{a=2}^M\nu(a)(1-\frac1a)$ is an even integer, and let $m$ and $m'$ contain each $a\in\{2,\dots,M\}$ exactly $\max(\nu(a),0)$, respectively $\max(-\nu(a),0)$, times. For every integer $g\ge\max(0,-s/2)$ with $2g-2+\sum_{x\in m}(1-\frac1x)>0$, the signatures $(g;m)$ and $(g+\frac s2;m')$ are those of orbifolds in $\Sig_{\le M}$; they are different, the orbifolds share $c_1,\dots,c_{M-1}$, and $c_M(g;m)-c_M(g+\frac s2;m')=(-1)^MC\,a_{M-2}\ne0$, where $a_l=|B_{2l+2}|/(2(l+1)!\,(2l+1))$ is the leading coefficient of $p_l$ (Lemma~\ref{lem:conepoly}). In particular $\max_{\Orb\in\Sig_{\le M}}\Kmult(\Orb;\Sig_{\le M})=M$.
\item Against all competitors: if $\Orb\in\Sig_{\le M}$ has area $A$ and $\Orb'\in\Sig$ shares $c_1,\dots,c_L$ with $\Orb$ for some $L\ge2$, every cone order of $\Orb'$ is at most $M\,(2\lfloor A/\pi\rfloor+8)^{1/(2L-3)}$; and
\[
  \Kmult(\Orb;\Sig)\ \le\ M+\Big\lceil\tfrac12\log\big(2\lfloor A/\pi\rfloor+8\big)\Big\rceil .
\]
\end{enumerate}
\end{theorem}

For $M=3$, $\nu(2)=-16$, $\nu(3)=9$ and $s=-2$, and the least choice gives $(1;3,\dots,3)$, with nine cone points, against $(0;2,\dots,2)$, with sixteen, both of area $12\pi$. For $M=4$, $(\nu(2),\nu(3),\nu(4))=(28,-27,8)$ gives $(0;2^{28},4^8)$ against $(1;3^{27})$, of area $36\pi$ (exponents are multiplicities). In both cases an exhaustive search shows that no pair of smaller area with orders at most $M$ shares $c_1,\dots,c_{M-1}$. In general the least choice need not have least area: for $M=2$ it gives $(1;2,2,2,2)$ against $(2;\,)$, of area $4\pi$, while $(0;2,2,2,2,2)$ and $(1;2)$, of area $\pi$, also share $c_1$; adding a common cone point to both sides of a pair keeps the shared invariants.

\begin{proof}
(i) Put $\nu(a)=\mult_m(a)-\mult_{m'}(a)$ for $2\le a\le M$, where $m,m'$ are the cone-order multisets. By Lemma~\ref{lem:sigdata}, $H_M(\Orb)=H_M(\Orb')$ gives $2g+n-R(m)=2g'+n'-R(m')$ and
\[
  0=\Psi_k(m)-\Psi_k(m')=\sum_{a=2}^M\nu(a)\,\psi_k(a)=\sum_{a=2}^M\frac{\nu(a)}a\big(y_a^k-1\big)\qquad(1\le k\le M-1),\qquad y_a=a^2 .
\]
The $(M-1)\times(M-1)$ matrix $(y_a^k-1)_{1\le k\le M-1,\ 2\le a\le M}$ is invertible: in the Vandermonde matrix $(y_a^k)_{0\le k\le M-1,\ 1\le a\le M}$ with $y_1=1$, subtracting the column $a=1$ from the others turns row $k=0$ into $(1,0,\dots,0)$, so expanding along that row shows that the determinant of our matrix is the Vandermonde determinant $\prod_{1\le a<b\le M}(y_b-y_a)\ne0$. Hence $\nu=0$, that is $m=m'$, and then $n=n'$, $R(m)=R(m')$ and the first equation give $g=g'$. The second statement follows, with Corollary~\ref{cor:sigarea}, since $\Sig_{\le M}\subset\Sig$.

(ii) The $w_a$, $1\le a\le M$, are the weights of Lagrange interpolation at the distinct nodes $y_1,\dots,y_M$: for every polynomial $F$ of degree at most $M-2$, $\sum_{a=1}^Mw_aF(y_a)$ is the coefficient of $y^{M-1}$ in the interpolating polynomial of $F$, which is $F$ itself, so the sum vanishes. Taking $F=y^k$ and $F=1$,
\[
  \sum_{a=2}^M\nu(a)\psi_k(a)=C\sum_{a=2}^Mw_a\big(y_a^k-1\big)=C\sum_{a=1}^Mw_ay_a^k-C\sum_{a=1}^Mw_a=0\qquad(1\le k\le M-2),
\]
the term $a=1$ vanishing because $y_1^k=1$. So $\Psi_k(m)=\Psi_k(m')$ for $k\le M-2$. A positive integer $C$ as in the statement exists, since the $a\,w_a$ are rational, and every $\nu(a)$ is nonzero, as $w_a\ne0$; so $m\ne m'$ and the signatures differ. With $g'=g+\frac s2$ and $n,n'$ the sizes of $m,m'$,
\[
  \big(2g'+n'-R(m')\big)-\big(2g+n-R(m)\big)=s-\sum_{a=2}^M\nu(a)\Big(1-\frac1a\Big)=0,
\]
so the two signatures have the same area $2\pi(2g-2+\sum_{x\in m}(1-\frac1x))$, which is positive by the choice of $g$; so both are hyperbolic, the second because the areas are equal, and $g'\ge0$ as $g\ge-s/2$. Lemma~\ref{lem:sigdata} gives equal $c_1,\dots,c_{M-1}$. For $c_M$, the terms $\alpha_{M-1}\Area/4\pi$ cancel and, by \eqref{eq:bl} and $p_{M-2}(1)=0$,
\[
  c_M(g;m)-c_M(g';m')=\sum_{a=2}^M\nu(a)\,b_{M-2}(a)=(-1)^MC\sum_{a=1}^Mw_a\,p_{M-2}(a),
\]
and since $p_{M-2}$ is a polynomial of degree $M-1$ in $y=a^2$ with leading coefficient $a_{M-2}$ (Lemma~\ref{lem:conepoly}), the Lagrange sum is $a_{M-2}$. Finally $\Kmult\ge M$ for these orbifolds, and $\Kmult\le M$ by part~(i).

(iii) The areas are equal, so Lemma~\ref{lem:sigdata} applies: the paddings $U\supseteq m$ and $V\supseteq m'$ by $1$s satisfy $P_{2L-3}(U)=P_{2L-3}(V)$, every element of $U$ is at most $M$, and $|U|\le|U|+|V|\le2\lfloor A/\pi\rfloor+8$ (proof of Corollary~\ref{cor:sigarea}). For an order $v$ of $\Orb'$, $v^{2L-3}\le P_{2L-3}(V)=P_{2L-3}(U)\le|U|M^{2L-3}$, which is the first claim. Put $\lambda=\log(2\lfloor A/\pi\rfloor+8)$, $k=\lceil\lambda/2\rceil\ge2$ (as $\lambda\ge\log8$) and $L=M+k$. With $y=(k+1)/M$, $\log(1+y)\ge2y/(2+y)$ gives
\[
  (2L-3)\log\frac{L+1}M\ \ge\ \frac{2(2M+2k-3)(k+1)}{2M+k+1}\ >\ 2k\ \ge\ \lambda,
\]
the middle inequality being equivalent to $2M+(k-3)(k+1)>0$. Hence $M(2\lfloor A/\pi\rfloor+8)^{1/(2L-3)}<L+1$. So if $\Orb'$ shares $c_1,\dots,c_L$ with $\Orb$, all its orders are at most $L$, both orbifolds lie in $\Sig_{\le L}$, and part~(i) with $L$ in place of $M$ gives $\sigma(\Orb')=\sigma(\Orb)$.
\end{proof}

\begin{corollary}[The divergence needs large cone orders]\label{msep:cor}
\leavevmode
\begin{enumerate}[label=(\alph*)]
\item If two orbifolds in $\Sig$ with different signatures share $c_1,\dots,c_L$, $L\ge2$, then one of them has a cone point of order at least $L+1$.
\item If $\Orb\in\Sig$ has area $A$, largest cone order $M\ge2$ and $\Kmult(\Orb;\Sig)\ge L+1$, then $M\ge L+1-\lceil\frac12\log(2\lfloor A/\pi\rfloor+8)\rceil$.
\item Let $f_M(A)=\max\{\Kmult(\Orb;\Sig):\Orb\in\Sig_{\le M},\ \Area(\Orb)\le A\}$. Then $M\le f_M(A)\le M+\lceil\frac12\log(2\lfloor A/\pi\rfloor+8)\rceil$ once $A$ is at least the area of the pair in~(ii), while $f(A)=\sup_Mf_M(A)$ is at least $\lfloor\sqrt{(A/2\pi-1)/3}\rfloor+2$ (Theorem~\ref{thm:growth}(a)).
\end{enumerate}
\end{corollary}

\begin{proof}
(a) If all orders of both were at most $L$, both would lie in $\Sig_{\le L}$, and part~(i) with $L$ in place of $M$ would give equal signatures. (b) is part~(iii). (c) The upper bound is part~(iii), and the lower bound is part~(ii), since $\Kmult(\Orb;\Sig)\ge\Kmult(\Orb;\Sig_{\le M})$.
\end{proof}
