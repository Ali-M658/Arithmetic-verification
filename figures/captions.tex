% F1 (Introduction) -- figures/out/F1.pdf
\newcommand{\figcapOne}{One triangle of $\mathcal O(2,8,8)$ (a) and of $\mathcal O(3,3,12)$ (b), in schematic, not
isometric, shapes, coloured by $4\pi t\,\mathfrak h_t(x,x)$ at $t=0.02$, with $\mathfrak h_t$ the heat kernel. The two
share their first two heat invariants, yet heat lingers differently: the value is near $1$ at smooth
points and approaches the order at each cone point, reaching $8$ or $12$ at the tips of largest order.}

% F2 (Section 5) -- figures/out/F2.pdf
\newcommand{\figcapTwo}{Exact tilings of the hyperboloid by the triangle groups of $\mathcal O(2,8,8)$ (a) and
$\mathcal O(3,3,12)$ (b), in an oblique orthographic view, so the circular rim appears as an ellipse; the
coloured triangle and its pale mirror image form one copy of the orbifold. The triangles differ in shape but have equal area, $\pi/4$, and equal order sum, all that
the first two heat invariants record.}

% F3 (Section 3) -- figures/out/F3.pdf
\newcommand{\figcapThree}{The mirror argument of Theorem~\ref{thm:sigsep}: discs mark $X^*=U^*\uplus(-V^*)$, in (a) in the
colour of each orbifold, in (b) black for $U^*$ (with a padding $1$) and grey for $-V^*$, and rings $-X^*$, which would coincide, as for an orbifold against itself (lower row of (a)), if a pair
shared $c_1,\dots,c_L$ with $|X^*|\le2L$. The pairs $(2,8,8)$, $(3,3,12)$ (a) and $(1;15)$,
$(0;3,3,5,5)$ (b) have $|X^*|=6$ and share exactly $c_1,c_2$.}

% F4 (Section 3) -- figures/out/F4.pdf
\newcommand{\figcapFour}{Dots: the largest $K_{\mathrm{mult}}$ on each of the $525$ complete equal-area classes
of Section~\ref{sec:pte}, the split disc the minimal pair; diamonds and squares: the pairs of
Table~\sref{tab:pairs}, each giving $K_{\mathrm{mult}}\ge L+1$. The open growth question lies between the bounds of
Corollary~\ref{cor:sigarea} (solid) and the guarantee of Theorem~\ref{thm:growth}(a), for $s\ge4$ (dashed).}

% F5 (Sections 4 and 7) -- figures/out/F5.pdf
\newcommand{\figcapFive}{(a) $D(t)=\mathrm Z_{(2,8,8)}(t)-\mathrm Z_{(3,3,12)}(t)$ from the computed spectra
(heavy) follows the cone-point term (dashed), and for small $t$ its divergent expansion (grey), until
closed geodesics take over beyond $t\approx0.1$. (b) In the family $(0;3,3,3,3)$,
$|\mathrm Z_0-\mathrm Z_\vartheta|$ for the seven $\vartheta>0$ (shorter systole darker) matches the predicted
geodesic term (thin dashed) above the error budget (shaded); circles mark its emergence.}

% F6 (Section 4) -- figures/out/F6.pdf
\newcommand{\figcapSix}{(a) Schematic, not isometric, shapes of four members of the family $(0;3,3,3,3)$ with
their shortest closed geodesics, of lengths $2.634$ to $0.694$; (b) the eigenvalues below $120$ at
eight moduli $\vartheta$ (square-root scale), $\lambda_1$ heavy. All members share every heat invariant, yet the
spectrum moves: $\lambda_1$ falls from $4.122$ to $0.435$.}

% F7 (Section 5) -- figures/out/F7.pdf
\newcommand{\figcapSeven}{Reciprocal sums $R$ of the hyperbolic triads of sum $S$ and least order
$p=2,\dots,8$ (bands defined at integers $S$, light to dark; logarithmic axes), with dots at the first overlap of
adjacent bands and circles at their first collision. The two coincide only for $p=2,4$, and the
first collision of all is $(2,8,8)$, $(3,3,12)$ (split disc).}

% F8 (Section 6) -- figures/out/F8.pdf
\newcommand{\figcapEight}{Recovery error of the cone orders against a worst-case error $\delta$ in every heat
invariant for $(2,3,7)$ (solid), $(2,8,8)$ (heavy) and $(4,4,4)$ (dashed; dotted: errors from real
multisets), with slopes $1,\frac12,\frac13,\frac12$. Diamonds mark the threshold
$\delta_{\rm cert}$ certified in the supplement, nearly sharp for $(2,8,8)$ and $(4,4,4)$; the thin line is the
rounding limit $\frac12$.}
