% F1 (Introduction) -- figures/out/F1.pdf
\newcommand{\figcapOne}{Stylised renderings, not isometric embeddings, of the pillows
$\mathcal O(2,8,8)$ (a) and $\mathcal O(3,3,12)$ (b), at one scale and seen from above (only the
upper sheet shows), coloured by $4\pi t\,\mathfrak h_t(x,x)$, with $\mathfrak h_t$ the heat kernel, at $t=0.02$ on one logarithmic scale,
darker meaning larger. Away from the cone points the value is $1-t/3+O(t^2)$, and at a cone point
of order $m$ it rises to $m$: the sharp dark tips are the points of order $8$ (a) and $12$ (b), the
blunt corners those of order $2$ and $3$. The two pillows have the same area and the same
$c_1,c_2$.}

% F2 (Section 5) -- figures/out/F2.pdf
\newcommand{\figcapTwo}{The tilings of the upper sheet of the hyperboloid model by the triangle
groups $\Gamma(2,8,8)$ (a) and $\Gamma(3,3,12)$ (b), drawn exactly in an oblique orthographic view
on the cap of points within distance $2.4$ of the vertex of $T_{p,q,r}$ with angle $\pi/p$. The
triangle $T_{p,q,r}$ (deep shade of the pillow's hue) and its mirror image in a side (pale shade)
form one copy of the pillow. The other tiles alternate between white and grey.}

% F3 (Section 3) -- figures/out/F3.pdf
\newcommand{\figcapThree}{The mirror argument of Theorem~\ref{thm:sigsep}: discs mark $X^*=U^*\uplus(-V^*)$ on
the real axis (stacked by multiplicity), rings mark $-X^*$, and sharing $c_1,\dots,c_L$ with
$|X^*|\le 2L$ forces $X^*=-X^*$, hence equal signatures. (a) $(2,8,8)$ (dark) against $(3,3,12)$
(light): $|X^*|=6$ and exactly $c_1,c_2$ are shared; below, the symmetric configuration that
sharing $c_3$ as well would force. (b) $(1;15)$ (black) against $(0;3,3,5,5)$ (grey), with
$U=\{1,15\}$, i.e.\ $15$ padded by one order~$1$: $|V|-|U|=2=2(g-g')$, and exactly $c_1,c_2$ are
shared.}

% F4 (Section 3) -- figures/out/F4.pdf
\newcommand{\figcapFour}{Dots: for each of the $525$ complete area classes with
$s=\mathrm{Area}/2\pi\le 7/5$, the largest $K_{\mathrm{mult}}(\mathcal O;\mathrm{Sig})$ (the least
$k$ such that $c_1,\dots,c_k$ determine the signature of $\mathcal O$); the split disc (half dark,
half light) is the class of $(2,8,8)$ and $(3,3,12)$. Solid steps: the upper bound
$\lfloor 2s\rfloor+4$. Dashed steps, from $s=3$ (the hypothesis $A\ge 6\pi$ of Corollary~\ref{cor:siggrowth}):
the lower bound $\lfloor\log_4(s+1)\rfloor+2$ for $\max_{\mathrm{Area}\le 2\pi s}K_{\mathrm{mult}}$;
squares: the pairs of Theorem~\ref{thm:signonuniform}, which share $c_1,\dots,c_L$ ($L=2,\dots,6$).}

\newcommand{\figcapFive}{(a) $D(t)=Z_{(2,8,8)}(t)-Z_{(3,3,12)}(t)$ from the computed spectra

% F6 (Sections 4.2 and 7.2) -- figures/out/F6.pdf
\newcommand{\figcapSix}{(a) Four members $\mathcal O(\vartheta)$ of the family $(0;3,3,3,3)$,
$\vartheta=0,0.8,1.6,2.8$ from left to right (stylised, not isometric, at one scale; the four

% F7 (Section 5) -- figures/out/F7.pdf
\newcommand{\figcapSeven}{The least-order strata of hyperbolic triads, $p=2,\dots,8$ (light to
dark grey; logarithmic axes): at each sum $S$ the values of $R$ in stratum $p$ lie between
$R^-_{S,p}$ and $R^+_{S,p}$ (joined across integer $S$). A dot marks the first overlap $S^*(p)$ of

% F8 (Section 6) -- figures/out/F8.pdf
\newcommand{\figcapEight}{Recovery error of the cone orders against an error $\delta$ in every
heat coefficient (worst signs) for $(2,3,7)$ (solid), $(2,8,8)$ (heavy, dark) and $(4,4,4)$
(dashed), and for $(4,4,4)$ under errors that come from real multisets (dotted); the slopes are
